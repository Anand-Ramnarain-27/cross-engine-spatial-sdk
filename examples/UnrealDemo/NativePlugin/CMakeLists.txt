if(NOT WIN32)
    message(STATUS "SpatialUnrealPlugin targets Windows x86_64 Unreal builds only; skipping.")
    return()
endif()

# Same reasoning as examples/UnityDemo/NativePlugin: one self-contained DLL,
# spatial::sdk linked straight in (built static by default).
add_library(SpatialUnrealPlugin SHARED
    src/SpatialUnrealPlugin.h
    src/SpatialUnrealPlugin.cpp
    src/UnrealCoordinateConversion.h
    ${CMAKE_SOURCE_DIR}/examples/common/ManagedMeshRenderer.h
    ${CMAKE_SOURCE_DIR}/examples/common/ManagedMeshRenderer.cpp
)

target_link_libraries(SpatialUnrealPlugin PRIVATE spatial::sdk)
target_compile_features(SpatialUnrealPlugin PRIVATE cxx_std_20)
target_include_directories(SpatialUnrealPlugin PRIVATE src ${CMAKE_SOURCE_DIR}/examples/common)

if(MSVC)
    target_compile_options(SpatialUnrealPlugin PRIVATE /W4)
    if(SPATIAL_SDK_WARNINGS_AS_ERRORS)
        target_compile_options(SpatialUnrealPlugin PRIVATE /WX)
    endif()
endif()

# Stages the DLL, its import lib, and the public header into the Unreal
# plugin's ThirdParty layout (Source/ThirdParty/SpatialUnrealPlugin/), which
# is what SpatialSDKPlugin.Build.cs references. Not committed to source
# control (see .gitignore) — regenerated by this target like any other
# build output.
set(SPATIAL_UNREAL_PLUGIN_DEST "${CMAKE_CURRENT_SOURCE_DIR}/../UnrealProject/Plugins/SpatialSDKPlugin/Source/ThirdParty/SpatialUnrealPlugin")
add_custom_command(TARGET SpatialUnrealPlugin POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${SPATIAL_UNREAL_PLUGIN_DEST}/include"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${SPATIAL_UNREAL_PLUGIN_DEST}/lib"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CMAKE_CURRENT_SOURCE_DIR}/src/SpatialUnrealPlugin.h"
        "${SPATIAL_UNREAL_PLUGIN_DEST}/include/SpatialUnrealPlugin.h"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "$<TARGET_FILE:SpatialUnrealPlugin>"
        "${SPATIAL_UNREAL_PLUGIN_DEST}/lib/$<TARGET_FILE_NAME:SpatialUnrealPlugin>"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "$<TARGET_LINKER_FILE:SpatialUnrealPlugin>"
        "${SPATIAL_UNREAL_PLUGIN_DEST}/lib/$<TARGET_LINKER_FILE_NAME:SpatialUnrealPlugin>"
    COMMENT "Staging SpatialUnrealPlugin.dll/.lib/header into SpatialSDKPlugin/Source/ThirdParty"
)
