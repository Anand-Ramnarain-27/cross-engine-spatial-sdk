if(NOT WIN32)
    message(STATUS "SpatialUnityPlugin targets Windows x86_64 Unity builds only; skipping.")
    return()
endif()

# Unity's native plugin loader expects one self-contained DLL, so this links
# spatial::sdk (built static by default — see SPATIAL_SDK_BUILD_SHARED)
# straight in rather than shipping it alongside as a second DLL.
add_library(SpatialUnityPlugin SHARED
    src/SpatialUnityPlugin.h
    src/SpatialUnityPlugin.cpp
    ${CMAKE_SOURCE_DIR}/examples/common/ManagedMeshRenderer.h
    ${CMAKE_SOURCE_DIR}/examples/common/ManagedMeshRenderer.cpp
)

target_link_libraries(SpatialUnityPlugin PRIVATE spatial::sdk)
target_compile_features(SpatialUnityPlugin PRIVATE cxx_std_20)
target_include_directories(SpatialUnityPlugin PRIVATE src ${CMAKE_SOURCE_DIR}/examples/common)

if(MSVC)
    target_compile_options(SpatialUnityPlugin PRIVATE /W4)
    if(SPATIAL_SDK_WARNINGS_AS_ERRORS)
        target_compile_options(SpatialUnityPlugin PRIVATE /WX)
    endif()
endif()

# Drop the built DLL straight into the Unity project's plugin folder so
# opening UnityProject in the Editor after a build picks it up immediately.
# The DLL itself isn't committed to source control (see .gitignore) — it's a
# build artifact regenerated by this target, same as any other example.
set(SPATIAL_UNITY_PLUGIN_DEST "${CMAKE_CURRENT_SOURCE_DIR}/../UnityProject/Assets/Plugins/x86_64")
add_custom_command(TARGET SpatialUnityPlugin POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${SPATIAL_UNITY_PLUGIN_DEST}"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "$<TARGET_FILE:SpatialUnityPlugin>"
        "${SPATIAL_UNITY_PLUGIN_DEST}/$<TARGET_FILE_NAME:SpatialUnityPlugin>"
    COMMENT "Copying SpatialUnityPlugin.dll into UnityProject/Assets/Plugins/x86_64"
)
